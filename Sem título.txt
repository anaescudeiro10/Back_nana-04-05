
CREATE TABLE USUARIOS (
    id_usuario INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL,
    tipo VARCHAR(20),

    conta_confirmada BOOLEAN DEFAULT FALSE, 
    data_cadastro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);


CREATE TABLE CONFIRMAR_CODIGO (
    id_confirmacao INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_usuario INTEGER NOT NULL, 
    codigo VARCHAR(6) NOT NULL,
    data_geracao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    utilizado BOOLEAN DEFAULT FALSE,
    CONSTRAINT FK_CONFIRM_USUARIO FOREIGN KEY (id_usuario) 
        REFERENCES USUARIOS(id_usuario) ON DELETE CASCADE
);



CREATE TABLE HISTORICO_SENHAS (
    id_historico INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    senha_antiga VARCHAR(255) NOT NULL,
    data_alteracao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT FK_HISTORICO_USUARIO FOREIGN KEY (id_usuario) 
        REFERENCES USUARIOS(id_usuario) ON DELETE CASCADE
);

CREATE TABLE COLECOES (
    id_colecao INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao BLOB SUB_TYPE TEXT
);

CREATE TABLE CORES (
    id_cor INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(50) NOT NULL,
    codigo_hex VARCHAR(7)
);

CREATE TABLE TAMANHOS (
    id_tamanho INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    numero INTEGER NOT NULL
);

CREATE TABLE PRODUTOS (
    id_produto INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao BLOB SUB_TYPE TEXT,
    preco DECIMAL(10,2) NOT NULL,
    id_colecao INTEGER,
    CONSTRAINT FK_PRODUTOS_COLECAO FOREIGN KEY (id_colecao) REFERENCES COLECOES(id_colecao)
);

CREATE TABLE CLIENTES (
    id_client INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    endereco BLOB SUB_TYPE TEXT,
    CONSTRAINT UNQ_CLIENTE_USUARIO UNIQUE (id_usuario),
    CONSTRAINT FK_CLIENTES_USUARIO FOREIGN KEY (id_usuario) REFERENCES USUARIOS(id_usuario)
);

CREATE TABLE GRADE (
    id_grade INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_produto INTEGER NOT NULL,
    id_cor INTEGER NOT NULL,
    id_tamanho INTEGER NOT NULL,
    quantidade_estoque INTEGER NOT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    CONSTRAINT FK_GRADE_PRODUTO FOREIGN KEY (id_produto) REFERENCES PRODUTOS(id_produto),
    CONSTRAINT FK_GRADE_COR FOREIGN KEY (id_cor) REFERENCES CORES(id_cor),
    CONSTRAINT FK_GRADE_TAMANHO FOREIGN KEY (id_tamanho) REFERENCES TAMANHOS(id_tamanho)
);

CREATE TABLE PEDIDOS (
    id_pedido INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_cliente INTEGER NOT NULL,
    id_vendedor INTEGER NOT NULL,
    data_pedido DATE DEFAULT CURRENT_DATE NOT NULL,
    status VARCHAR(20) NOT NULL,
    valor_total DECIMAL(10,2) NOT NULL,
    CONSTRAINT FK_PEDIDOS_CLIENTE FOREIGN KEY (id_cliente) REFERENCES CLIENTES(id_client),
    CONSTRAINT FK_PEDIDOS_VENDEDOR FOREIGN KEY (id_vendedor) REFERENCES USUARIOS(id_usuario)
);

CREATE TABLE ITENS_PEDIDO (
    id_item INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_pedido INTEGER NOT NULL,
    id_grade INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,
    CONSTRAINT FK_ITENS_PEDIDO_PEDIDO FOREIGN KEY (id_pedido) REFERENCES PEDIDOS(id_pedido),
    CONSTRAINT FK_ITENS_PEDIDO_GRADE FOREIGN KEY (id_grade) REFERENCES GRADE(id_grade)
);

    ID_HISTORICO INTEGER NOT NULL PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    SENHA_ANTIGA VARCHAR(255) NOT NULL,
    DATA_ALTERACAO TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (ID_USUARIO) REFERENCES USUARIOS(ID_USUARIO) ON DELETE CASCADE
);

/* Generator para ID do histórico (comum no Firebird) */
CREATE GENERATOR GEN_HISTORICO_ID;
SET GENERATOR GEN_HISTORICO_ID TO 0;

/* Trigger para auto-incremento no histórico */
SET TERM ^ ;
CREATE TRIGGER BI_HISTORICO_ID FOR HISTORICO_SENHAS
ACTIVE BEFORE INSERT POSITION 0
AS
BEGIN
  IF (NEW.ID_HISTORICO IS NULL) THEN
    NEW.ID_HISTORICO = NEXT VALUE FOR GEN_HISTORICO_ID;
END^
SET TERM ; ^

COMMIT;
